function newGoal(){
 openModal("Nova meta",goalForm()+`<button type="button" class="primary modalSave">💾 Salvar meta</button>`,()=>{
   let type=document.getElementById("goalType")?.value||"Confecção";
   let scope=document.getElementById("goalScope")?.value||"";
   if(!scope){alert("Selecione o local, facção ou pessoa.");return}
   db.goals.push({
     id:Date.now(),scopeType:type,scopeName:scope,
     peopleCount:Math.max(1,+document.getElementById("goalPeople")?.value||1),
     goalPerPerson:+document.getElementById("goalPerPerson")?.value||0,
     targetValue:(Math.max(1,+document.getElementById("goalPeople")?.value||1))*(+document.getElementById("goalPerPerson")?.value||0),
     cadence:document.getElementById("goalCadence")?.value||"Semanal",
     note:document.getElementById("goalNote")?.value||""
   });
   closeModal();save();renderGoals();
 });
}
function editGoal(id){
 let g=(db.goals||[]).find(x=>+x.id===+id);if(!g)return;
 openModal("Editar meta",goalForm(g)+`<button type="button" class="primary modalSave" onclick="modalSave()">💾 Salvar alterações</button>`,()=>{
   Object.assign(g,{
     scopeType:document.getElementById("goalType")?.value||g.scopeType,
     scopeName:document.getElementById("goalScope")?.value||g.scopeName,
     peopleCount:Math.max(1,+document.getElementById("goalPeople")?.value||1),
     goalPerPerson:+document.getElementById("goalPerPerson")?.value||0,
     targetValue:(Math.max(1,+document.getElementById("goalPeople")?.value||1))*(+document.getElementById("goalPerPerson")?.value||0),
     cadence:document.getElementById("goalCadence")?.value||"Semanal",
     note:document.getElementById("goalNote")?.value||""
   });
   closeModal();save();renderGoals();
 });

function deleteGoal(id){
 if(!confirm("Excluir esta meta?"))return;
 db.goals=(db.goals||[]).filter(x=>+x.id!==+id);
 save();renderGoals();
}
