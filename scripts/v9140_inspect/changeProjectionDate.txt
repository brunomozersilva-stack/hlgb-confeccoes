function changeProjectionDate(orderId){
 let o=(db.orders||[]).find(x=>+x.id===+orderId);if(!o)return;
 let items=projectionItemsForOrder(o);
 let rows=items.map((it,i)=>`<tr>
   <td>${esc(it.name)}</td>
   <td>${it.qty.toLocaleString("pt-BR")}</td>
   <td>${money(it.value||0)}</td>
   <td><input id="projItemDate_${i}" type="date" value="${it.date||o.projectionDeliveryDate||o.date||isoDate(new Date())}"></td>
 </tr>`).join("");
 openModal("Programar projeção de entrega por produto",`
   <div class="grid">
    <div class="field"><label>Pedido</label><input value="#${typeof displayOrderNumber==="function"?displayOrderNumber(o):o.id}" disabled></div>
    <div class="field"><label>Cliente</label><input value="${esc(o.client||"")}" disabled></div>
    <div class="field"><label>Data prometida ao cliente</label><input value="${o.date?fmtDate(o.date):"-"}" disabled></div>
   </div>
   <div class="sub" style="margin:10px 0">Defina uma data diferente para cada modelo, se precisar. O pedido continua sendo um só.</div>
   <div style="overflow:auto"><table><thead><tr><th>Produto</th><th>Peças</th><th>Valor projetado</th><th>Data projetada de saída</th></tr></thead><tbody>${rows}</tbody></table></div>
   <button type="button" class="primary modalSave">💾 Salvar projeção por produto</button>`,()=>{
     items.forEach((it,i)=>{
       let d=document.getElementById(`projItemDate_${i}`)?.value||"";
       if(d)saveProjectionItemDate(o,it.key,d);
     });
     closeModal();save();
     let allDates=projectionItemsForOrder(o).map(x=>x.date).filter(Boolean).sort();
     if(allDates.length){
       let base=new Date(allDates[0]+"T12:00:00"),mon=mondayOf(base),sun=new Date(mon);sun.setDate(sun.getDate()+6);
       let s=document.getElementById("projectionStart"),e=document.getElementById("projectionEnd");
       if(s)s.value=isoDate(mon);if(e)e.value=isoDate(sun);
     }
     renderProjection();
   });
}

function finalizeSelectedProjection(){
 let keys=[...document.querySelectorAll(".projectionSelect:checked")].map(x=>x.value);
 if(!keys.length){alert("Selecione pelo menos um produto.");return}
 let selected=keys.map(v=>{
   let [oid,key]=String(v).split("::");
   let o=(db.orders||[]).find(x=>+x.id===+oid);if(!o)return null;
   let item=projectionItemsForOrder(o).find(i=>String(i.key)===key);if(!item)return null;
   return {o,item};
 }).filter(Boolean);
 let sourceClients=[...new Set(selected.map(x=>x.o.client||"Sem cliente"))];
 if(sourceClients.length!==1){alert("Selecione produtos de um mesmo pedido/cliente de origem por vez.");return}

   return `<tr>
     <td>${esc(x.item.name)}</td>
     <td>${x.item.qty.toLocaleString("pt-BR")}</td>

     <td><input class="projectionInvoiceUnit" data-index="${i}" type="number" min="0" step=".01" value="${unit.toFixed(2)}" oninput="recalcProjectionInvoiceModal()"></td>
   </tr>`;
 }).join("");
