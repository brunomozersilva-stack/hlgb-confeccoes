function ensureOrderNumbers(){
 if(!db.config)db.config={};
 db.orders=Array.isArray(db.orders)?db.orders:[];
 let used=new Set();
 let next=1;
 db.orders.slice().sort((a,b)=>(+a.id||0)-(+b.id||0)).forEach(o=>{
   if(o.orderNumber!=null && String(o.orderNumber).trim()!==""){
     used.add(String(o.orderNumber));
     let n=+o.orderNumber;if(Number.isFinite(n))next=Math.max(next,n+1);
   }else{
     while(used.has(String(next)))next++;
     o.orderNumber=next;
     used.add(String(next));
     next++;
   }
 });
 if(!db.config.nextOrderNumber || +db.config.nextOrderNumber<next)db.config.nextOrderNumber=next;
 try{persistDb()}catch(e){}
}
function displayOrderNumber(o){return o?.orderNumber!=null?String(o.orderNumber):String(o?.id||"")}
function nextAvailableOrderNumber(){
 ensureOrderNumbers();
 let n=Math.max(1,+db.config.nextOrderNumber||1);
 let used=new Set((db.orders||[]).map(o=>String(o.orderNumber)));
 while(used.has(String(n)))n++;
 return n;
}

 try{persistDb()}catch(e){}

}
