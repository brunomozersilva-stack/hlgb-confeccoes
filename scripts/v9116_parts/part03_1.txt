function hlgb916ReadAdHocLines(){return [...document.querySelectorAll('.adHocLine')].map(r=>{let productId=+r.querySelector('.adHocProduct')?.value||0,qty=Math.max(0,+r.querySelector('.adHocQty')?.value||0),price=Math.max(0,+r.querySelector('.adHocPrice')?.value||0);return {productId,qty,price,product:hlgb916ProductById(productId)}}).filter(x=>x.productId&&x.qty>0)}
function hlgb916RecalcAdHoc(){let lines=hlgb916ReadAdHocLines(),q=lines.reduce((a,x)=>a+x.qty,0),v=lines.reduce((a,x)=>a+x.qty*x.price,0);let el=document.getElementById('adHocPreview');if(el)el.innerHTML=`<b>${q.toLocaleString('pt-BR')} peças</b> · ${money(v)}`}
