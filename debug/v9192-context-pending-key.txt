    m.set(String(r.entity_id),{
      data,deleted_at:r.deleted_at||null,revision:+r.revision||1,updated_at:r.updated_at||"",updated_by:r.updated_by||null
    });
    if(String(r.updated_at||"")>max)max=String(r.updated_at||"");
  });
  hlgbRecordSnapshots[module]=m;if(max)hlgbRecordLastSeen[module]=max;
}

function hlgbRecordPendingStore(){
  if(!hlgbRecordReady||cloudApplying||cloudBootstrapping)return;
  try{
    const pending={at:Date.now(),email:normEmail(cloudUser?.email||""),modules:{}};let any=false;
    for(const module of HLGB_RECORD_MODULES){
      // Não deixa um usuário secundário ficar eternamente “pendente” por mudanças
      // internas de módulos que ele não tem permissão para gravar.
      if(!hlgbRecordCanWrite(module))continue;
      const snap=hlgbRecordSnapshots[module]||new Map(),local=hlgbRecordLocalMap(module),ops=[];
      for(const [id,item] of local){
        const old=snap.get(id);
        if(!old||old.deleted_at||!cloudEq(old.data,item)){ops.push({id,data:hlgbRecordClone(item),deleted:false});any=true}
      }
      // v91.26: ausência local nunca gera exclusão automática na nuvem.
      if(ops.length)pending.modules[module]=ops;
    }
    if(any)localStorage.setItem(HLGB_RECORD_PENDING_KEY,JSON.stringify(pending));else hlgbRecordPendingClear();
  }catch(e){console.warn("HLGB record pending",e)}
}
async function hlgbRecordLoadDirect(since=null){
  // Carga em pacotes pequenos: evita que o Safari interrompa uma resposta grande.
  // O RPC é SECURITY INVOKER, exige sessão autenticada e continua protegido por RLS.
  const pageSize=250,rows=[];let serverTime="";
  try{
    for(let offset=0;offset<12000;offset+=pageSize){
      let packet=await cloudRequest("rpc/hlgb_load_records_page",{method:"POST",body:JSON.stringify({p_offset:offset,p_limit:pageSize,p_since:since||null})});
      if(Array.isArray(packet)&&packet.length===1)packet=packet[0];
      const page=Array.isArray(packet?.rows)?packet.rows:[];
      if(!packet||typeof packet!=="object")throw new Error("O pacote paginado não respondeu corretamente.");
      rows.push(...page);serverTime=String(packet.server_time||serverTime||"");
      if(packet.has_more!==true||page.length<pageSize)break;
    }
  }catch(rpcPageError){
    console.warn("HLGB: RPC paginado indisponível; tentando leitura REST paginada.",rpcPageError);
    rows.length=0;
    for(let offset=0;offset<12000;offset+=pageSize){
      let filter=since?`&updated_at=gte.${encodeURIComponent(since)}`:"";
      let page=await cloudRequest(`hlgb_records?select=module,entity_id,data,deleted_at,revision,updated_at,updated_by${filter}&order=updated_at.asc&limit=${pageSize}&offset=${offset}`);
      if(!Array.isArray(page))throw new Error("A leitura REST paginada não retornou uma lista de registros.");
      rows.push(...page);if(page.length<pageSize)break;
    }
  }
  const visibleModules=[...new Set(rows.map(r=>String(r?.module||"")).filter(m=>HLGB_RECORD_MODULES.includes(m)))];
  const modules=normEmail(cloudUser?.email)===normEmail(HLGB_OWNER_EMAIL)?HLGB_RECORD_MODULES.slice():visibleModules;
  return {modules,rows,server_time:serverTime||new Date().toISOString(),fallback:true};
}
async function hlgbRecordLoadBundle({preserveLocal=false,since=null}={}){
  const body={p_since:since||null};
  let bundle;
  const ownerLoading=normEmail(cloudUser?.email)===normEmail(HLGB_OWNER_EMAIL);
  if(ownerLoading){
    // O administrador lê todos os módulos. Dividir em páginas evita que Safari e
    // redes móveis interrompam o login ao receber mais de mil registros de uma vez.
    bundle=await hlgbRecordLoadDirect(since);
  }else{
    try{
      bundle=await cloudRequest("rpc/hlgb_load_bundle",{method:"POST",body:JSON.stringify(body)});
    }catch(primaryError){
      console.warn("HLGB: pacote principal indisponível; usando leitura paginada segura.",primaryError);
      bundle=await hlgbRecordLoadDirect(since);
    }
  }
  if(Array.isArray(bundle)&&bundle.length===1&&bundle[0]&&bundle[0].rows)bundle=bundle[0];
  if(!bundle||typeof bundle!=="object")bundle=await hlgbRecordLoadDirect(since);
  const readable=Array.isArray(bundle.modules)?bundle.modules.filter(m=>HLGB_RECORD_MODULES.includes(String(m))):[];
  const rows=Array.isArray(bundle.rows)?bundle.rows:[];
  readable.forEach(m=>hlgbRecordReadable.add(String(m)));
  if(!since){
    for(const module of readable){
      const moduleRows=rows.filter(r=>String(r?.module||"")===String(module));
      hlgbRecordSnapshotRows(module,moduleRows);
      const remoteActive=moduleRows.filter(r=>!r.deleted_at).map(r=>hlgbRecordRowValue(r,module));
      if(!preserveLocal){
        db[module]=remoteActive;
      }else{
        // HLGB v91.28 AUTHORITATIVE RECORD MERGE
        // Old versions could keep the legacy/local list (for example 44 products),
        // advance the cloud cursor to the newest timestamp and therefore never add
        // the missing records. Merge the full server snapshot before advancing.
        const local=hlgbRecordLocalMap(module);
        const pendingNow=hlgbRecordPendingRead();
        const pendingOps=Array.isArray(pendingNow?.modules?.[module])?pendingNow.modules[module]:[];
        const pendingIds=new Set(pendingOps.filter(op=>op&&op.deleted!==true).map(op=>String(op.id)));
        for(const row of moduleRows){
          const rid=String(row?.entity_id??"");
          if(!rid)continue;
          if(row.deleted_at){
            if(!pendingIds.has(rid))local.delete(rid);
            continue;
          }
          // Cloud wins unless this exact record has a confirmed local pending write.
          if(!pendingIds.has(rid))local.set(rid,hlgbRecordRowValue(row,module));
        }
        // Local-only new records are preserved; the pending queue will sync them.
        db[module]=[...local.values()];
      }
    }
  }else{
    for(const row of rows)hlgbApplyRecordRow(row);
  }
  for(const row of rows){
    const t=String(row?.updated_at||"");
    if(t>hlgbRecordGlobalLastSeen)hlgbRecordGlobalLastSeen=t;
  }
  const serverTime=String(bundle.server_time||"");
  if(!hlgbRecordGlobalLastSeen&&serverTime)hlgbRecordGlobalLastSeen=serverTime;
  return {readable,rows,serverTime};
}

async function hlgbRecordSaveWithRetry(module,id,data,deleted=false){
  let snap=hlgbRecordSnapshots[module]||new Map(),old=snap.get(String(id)),expected=+old?.revision||0;
  let result=await hlgbRecordRpcSave(module,id,hlgbRecordClone(data),expected,deleted);
  if(result?.applied===false){
    // Outra máquina editou exatamente o mesmo registro. Mescla campo a campo e tenta de novo.
    const remote=result.data;
    const base=old?.data;
    const merged=deleted?hlgbRecordClone(data):cloudMergeThreeWay(base,hlgbRecordClone(data),remote);
    result=await hlgbRecordRpcSave(module,id,merged,+result.revision||0,deleted);
    if(result?.applied===false)throw new Error(`Conflito ao salvar ${module}. Atualize e tente novamente.`);
    if(!deleted){
      const arr=Array.isArray(db[module])?db[module]:[],idx=arr.findIndex((x,i)=>hlgbRecordId(module,x,i)===String(id));
      if(idx>=0)arr[idx]=hlgbRecordClone(result.data||merged);
    }
  }
  if(!result)throw new Error(`Supabase não confirmou o salvamento de ${module}.`);
  const map=hlgbRecordSnapshots[module]||new Map();
  map.set(String(id),{data:hlgbRecordClone(result.data),deleted_at:result.deleted_at||null,revision:+result.revision||1,updated_at:result.updated_at||new Date().toISOString(),updated_by:result.updated_by||null});
  hlgbRecordSnapshots[module]=map;
  if(String(result.updated_at||"")>String(hlgbRecordLastSeen[module]||""))hlgbRecordLastSeen[module]=String(result.updated_at||"");
  return result;
}
async function hlgbNormalizedSyncModule(module){
  if(!hlgbRecordReady||!cloudAccessToken)return true;
  if(!hlgbRecordCanWrite(module))return true;
  const snap=hlgbRecordSnapshots[module]||new Map(),local=hlgbRecordLocalMap(module),jobs=[];
  for(const [id,item] of local){const old=snap.get(id);if(!old||old.deleted_at||!cloudEq(old.data,item))jobs.push({id,data:item,deleted:false})}
  // v91.26: nenhuma exclusão é inferida por diferença entre navegador e nuvem.
  for(const job of jobs)await hlgbRecordSaveWithRetry(module,job.id,job.data,job.deleted);
  return true;
}
