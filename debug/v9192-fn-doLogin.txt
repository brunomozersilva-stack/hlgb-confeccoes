function doLogin(){
 try{if(window.HLGB_SESSION_BOOT_WATCHDOG){clearTimeout(window.HLGB_SESSION_BOOT_WATCHDOG);window.HLGB_SESSION_BOOT_WATCHDOG=null}}catch(e){}
 try{var __hl=document.getElementById('sessionLoader');if(__hl)__hl.style.display='none'}catch(e){}
 const userEl=document.getElementById("loginUser");
 const passEl=document.getElementById("loginPass");
 const errorEl=document.getElementById("loginError");
 const loginEl=document.getElementById("loginScreen");
 const appEl=document.getElementById("appShell");
 const email=(userEl.value||"").trim();
 const password=passEl.value||"";
 if(!email||!password){errorEl.textContent="Informe seu e-mail e senha.";errorEl.style.display="block";return false}
 if(!email.includes("@")){
   errorEl.textContent="O acesso online usa e-mail. Este é um login antigo. Peça ao Administrador para ativá-lo em Sistema → Configurações → Usuários e permissões.";
   errorEl.style.display="block";return false;
 }
 errorEl.textContent="Entrando…";errorEl.style.display="block";
 (async()=>{
  try{
   await cloudSignIn(email,password);
   errorEl.textContent="Acesso confirmado. Carregando seus dados da nuvem…";errorEl.style.display="block";
   let state=await hlgbFastBootstrap9182();
   // Garante que o modo por registros esteja realmente ativo também no login manual.
   // Se a carga inicial falhou por qualquer motivo, tenta uma segunda vez antes de liberar o sistema.
   if(state!=="empty" && !hlgbRecordReady){
     const recordsOk=await hlgbEnsureRecordsOnlineAfterLogin();
     if(!recordsOk)throw new Error("Não foi possível iniciar o modo multiusuário."+(hlgbLastRecordLoadError?" Detalhe técnico: "+hlgbLastRecordLoadError:""));
   }
   if(normEmail(cloudUser?.email)===normEmail(HLGB_OWNER_EMAIL)){normalizeUserDirectoryForOwner();}
   let profile=null;
   try{profile=await ensureCloudUserAccessOnline()}catch(accessErr){
     // Um usuário pode tentar entrar imediatamente após o administrador cadastrá-lo.
     // Faz uma segunda leitura curta da nuvem antes de concluir que não está autorizado.
     if(/ainda não foi autorizado|não foi autorizado/i.test(String(accessErr?.message||""))){
       await new Promise(r=>setTimeout(r,900));
       state=await hlgbFastBootstrap9182();
       if(!hlgbRecordReady){
          const recordsOk=await hlgbEnsureRecordsOnlineAfterLogin();
          if(!recordsOk)throw new Error("Não foi possível carregar os registros multiusuário.");
        }
        profile=await ensureCloudUserAccessOnline();
     }else throw accessErr;
   }
   // v91.10: o login manual também precisa reconciliar o que já existia neste navegador
   // antes de a nuvem substituir db.products. Sem isso, a recuperação só rodava em sessão restaurada.
   await hlgbRecoverLocalOnlyProducts();
   cloudBootstrapping=false;
   db.userSessions=Array.isArray(db.userSessions)?db.userSessions:[];
   let sess={id:Date.now()+Math.random(),userId:profile.id,user:profile.name||cloudUser.email,login:cloudUser.email,loginAt:new Date().toISOString(),logoutAt:""};
   db.userSessions.push(sess);db.activeSessionId=sess.id;
   localSaveOnly();
   errorEl.textContent="";errorEl.style.display="none";
   loginEl.style.display="none";appEl.style.display="block";setTimeout(()=>window.hlgbMarkAuthenticated?.(),0);
   // v91.83: manutenção de backups fora do caminho crítico do login.
   setTimeout(()=>{
     try{
       Promise.resolve(hlgbPrepareStorage9178()).catch(e=>console.warn("HLGB v91.83: manutenção de backups adiada",e));
     }catch(e){console.warn("HLGB v91.83: manutenção de backups adiada",e)}
   },2500);
   // v91.12: erro de interface depois do login nunca pode apagar a sessão online.
   try{
     applyAccess();restoreNavGroups();
     // Renderizar depois de carregar a nuvem NÃO é uma alteração do usuário.
     {const prevApplying=cloudApplying;cloudApplying=true;try{renderAll()}finally{cloudApplying=prevApplying}}
   }catch(uiErr){
     console.error("HLGB v91.12 erro de interface após login; sessão preservada",uiErr);
     try{setCloudStatus("⚠️ Online · interface com aviso","warn")}catch(e){}
   }
   if(state==="empty"){
     if(confirm("Seu login online funcionou. A nuvem ainda está vazia. Deseja enviar agora os dados que já estão neste navegador para a nuvem?\n\nClique em OK para enviar os dados atuais. Clique em Cancelar se preferir importar primeiro o backup que você exportou.")){
       await cloudInitializeFromCurrentDb(true);
     }
   }
  }catch(e){
   console.error(e);
   const appAlreadyOpen=!!appEl && appEl.style.display==="block";
   const authenticated=!!cloudUser && (!!cloudAccessToken || !!cloudRefreshToken);
   // v91.12: se a autenticação já foi concluída e o painel já abriu, um erro posterior
   // é problema de interface/bootstrap, não de login. Preserva access + refresh token.
   if(appAlreadyOpen && authenticated){
     try{cloudStoreAuth()}catch(_e){}
     try{setCloudStatus("⚠️ Online · sessão preservada","warn")}catch(_e){}
     errorEl.textContent="";errorEl.style.display="none";
     return;
   }
   cloudAccessToken="";cloudRefreshToken="";cloudTokenExpiresAt=0;cloudUser=null;cloudReady=false;cloudRowId=null;cloudStoreAuth();
   let msg=e.message||"verifique e-mail e senha.";
   if(/invalid login credentials/i.test(msg))msg="E-mail ou senha incorretos. Se este usuário foi criado antes do sistema online, o Administrador precisa ativar o acesso online dele.";
   errorEl.textContent="Não foi possível entrar: "+msg;errorEl.style.display="block";passEl.value="";passEl.focus();
  }
 })();
 return false;
}
