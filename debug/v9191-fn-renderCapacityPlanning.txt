function renderCapacityPlanning(){
 let [start,end]=capacityDateRange();let rows=allProjectionRows().filter(r=>{let d=r.item.date||r.order.date||'';return(!start||d>=start)&&(!end||d<=end)});let locs=(db.productionLocations||[]).filter(x=>x.active!==false),usage={};locs.forEach(l=>usage[l.id]=0);
 rows.forEach(r=>assignmentsFor(r.order.id,r.item.key).forEach(a=>{if(usage[a.locationId]!=null)usage[a.locationId]+=+a.qty||0}));
 let total=rows.reduce((a,r)=>a+(+r.item.qty||0),0),assigned=rows.reduce((a,r)=>a+Math.min(+r.item.qty||0,assignmentsFor(r.order.id,r.item.key).reduce((s,x)=>s+(+x.qty||0),0)),0);
 let cards=document.getElementById('capacityCards');if(cards)cards.innerHTML=`<div class="card"><small>Peças previstas no período</small><strong>${total.toLocaleString('pt-BR')}</strong></div><div class="card"><small>Já distribuídas</small><strong>${assigned.toLocaleString('pt-BR')}</strong></div><div class="card"><small>Ainda sem local</small><strong>${Math.max(0,total-assigned).toLocaleString('pt-BR')}</strong></div><div class="card"><small>Locais ativos</small><strong>${locs.length}</strong></div>`;
 let lt=document.getElementById('capacityLocationTable');if(lt){let rr=locs.map(l=>{let used=usage[l.id]||0,cap=+l.weeklyCapacity||0,pct=cap?used/cap*100:0,cls=cap?(pct>=100?'bad':pct>=80?'warn':'ok'):'',status=cap?`${pct.toFixed(0)}%`:'Sem capacidade definida',models=capacityAllowedModels(l);return[esc(l.name),esc(l.capacityResponsible||'-'),used.toLocaleString('pt-BR'),cap?cap.toLocaleString('pt-BR'):'<b>Definir</b>',`<span class="badge ${cls}">${status}</span>`,cap?(cap-used).toLocaleString('pt-BR'):'-',esc(models.join(', ')||'Todos'),`<button class="secondary" onclick="openCapacityLocationEditor()">Editar capacidade</button>`]});lt.innerHTML=rr.length?table(['Local','Responsável','Carga prevista','Capacidade semanal','Ocupação','Saldo','Modelos permitidos','Ação'],rr):'<div class="empty">Cadastre locais de produção para usar o planejamento.</div>'}
 let ot=document.getElementById('capacityOrderTable');if(ot){let rr=rows.map(r=>{let aa=assignmentsFor(r.order.id,r.item.key),dist=aa.reduce((s,x)=>s+(+x.qty||0),0),places=aa.map(a=>{let l=locs.find(x=>+x.id===+a.locationId);return l?`${l.name}: ${(+a.qty||0).toLocaleString('pt-BR')}`:''}).filter(Boolean).join('<br>');let rem=Math.max(0,(+r.item.qty||0)-dist);return[`#${typeof displayOrderNumber==='function'?displayOrderNumber(r.order):r.order.id}`,esc(r.order.client||'-'),`<b>${esc(r.item.name)}</b>`,(+r.item.qty||0).toLocaleString('pt-BR'),dist.toLocaleString('pt-BR'),rem?`<span class="badge warn">${rem.toLocaleString('pt-BR')}</span>`:'<span class="badge ok">0</span>',places||'<span class="sub">Ainda sem local</span>',`<button class="primary" onclick="openCapacityAllocation(${r.order.id},'${String(r.item.key).replace(/'/g,"\\'")}')">Distribuir / dividir</button>`]});ot.innerHTML=rr.length?table(['Pedido','Cliente','Modelo','Total','Distribuído','Restante','Onde será produzido','Ação'],rr):'<div class="empty">Nenhum produto previsto neste período.</div>'}
}
