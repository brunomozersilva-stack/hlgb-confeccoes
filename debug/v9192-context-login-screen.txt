function hlgb928EnsureCloudCoverage(module){
  try{
    if(!HLGB_RECORD_MODULES.includes(module))return 0;
    const snap=hlgbRecordSnapshots?.[module];
    if(!(snap instanceof Map))return 0;
    const local=hlgbRecordLocalMap(module);let added=0;
    for(const [id,row] of snap.entries()){
      if(!row||row.deleted_at||local.has(String(id)))continue;
      const data=hlgbRecordClone(row.data);
      if(data&&typeof data==="object"){
        local.set(String(id),data);added++;
      }
    }
    if(added)db[module]=[...local.values()];
    return added;
  }catch(e){console.warn("HLGB v91.28 coverage",module,e);return 0}
}

function cutPriorityWeight(o){
  let p=String(o?.priority||"Padrão").normalize("NFD").replace(/[\u0300-\u036f]/g,"").toLowerCase();
  if(p.includes("urgentissimo"))return 0;
  if(p.includes("urgente"))return 1;
  return 2;
}

function prepareCutSchedule(){
  db.cuts=Array.isArray(db.cuts)?db.cuts:[];
  let today=isoDate(new Date()),changed=false;
  let orders=pedidosQuePrecisamDeCorte();
  orders.forEach(o=>{
    let c=obterCorteDoPedido(o);
    let w=cutPriorityWeight(o);
    // Urgente/urgentíssimo sem programação entra automaticamente no dia atual.
    if(!c.plannedCutDate && w<=1){c.plannedCutDate=today;c.autoScheduledPriority=true;changed=true}
    // Tudo que ficou para trás e não foi finalizado passa para o dia atual.
    if(String(c.status||"").toLowerCase()!=="finalizado" && c.plannedCutDate && c.plannedCutDate<today){
      c.carriedFromDate=c.carriedFromDate||c.plannedCutDate;
      c.lastCarriedFrom=c.plannedCutDate;
      c.plannedCutDate=today;
      c.rolloverCount=(+c.rolloverCount||0)+1;
      changed=true;
    }
  });
  if(changed){try{persistDb()}catch(e){console.error(e)}}
}
