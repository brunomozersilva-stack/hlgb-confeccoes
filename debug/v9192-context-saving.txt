async function cloudRefreshSession(force=false){
 if(cloudRefreshPromise)return cloudRefreshPromise;
 if(!cloudRefreshToken)throw new Error("Sua sessão expirou. Entre novamente para continuar.");
 if(!force&&cloudAccessToken&&cloudTokenExpiresAt>Date.now()+90000)return true;
 cloudRefreshPromise=(async()=>{
  // v91.74: algumas versões do Safari deixam a requisição de renovação pendurada
  // quando o refresh token guardado ficou obsoleto. Abortamos somente a chamada;
  // banco local, diário de pendências e dados da nuvem não são tocados.
  const controller=typeof AbortController!=="undefined"?new AbortController():null;
  const timeoutId=setTimeout(()=>{try{controller?.abort()}catch(e){}},8000);
  let res;
  try{
   res=await fetch(`${HLGB_SUPABASE_URL}/auth/v1/token?grant_type=refresh_token`,{
    method:"POST",headers:{"apikey":HLGB_SUPABASE_KEY,"Content-Type":"application/json"},
    body:JSON.stringify({refresh_token:cloudRefreshToken}),...(controller?{signal:controller.signal}:{})
   });
  }catch(e){
   if(e?.name==="AbortError")throw new Error("A sessão anterior não respondeu. Entre novamente para continuar.");
   throw e;
  }finally{clearTimeout(timeoutId)}
  let data=await res.json().catch(()=>({}));
  // 429 não significa sessão inválida: significa excesso temporário de refresh.
  // Se o JWT atual ainda é válido, continuamos usando-o e tentamos renovar depois.
  if(res.status===429){
   if(cloudAccessToken&&cloudTokenExpiresAt>Date.now()+5000)return true;
   throw new Error(data.error_description||data.msg||data.message||"Muitas tentativas de renovação da sessão. Aguarde alguns segundos.");
  }
  if(!res.ok||!data.access_token){
   cloudAccessToken="";cloudRefreshToken="";cloudTokenExpiresAt=0;cloudStoreAuth();
   throw new Error(data.error_description||data.msg||data.message||"Sua sessão expirou. Entre novamente.");
  }
  cloudApplyAuth(data);setCloudStatus("☁️ Online","ok");return true;
 })();
 try{return await cloudRefreshPromise}finally{cloudRefreshPromise=null}
}

async function hlgbPullNormalizedCoreChanges(force=false){
  if(!hlgbRecordReady||hlgbRecordPulling||!cloudAccessToken||document.hidden)return false;
  // Realtime continua sendo o caminho principal, mas mantemos UMA leitura em lote como segurança.
  // A chamada legada ocorre com frequência, porém este limitador permite no máximo 1 pacote/8 s,
  // em vez das dezenas de consultas por módulo que existiam nas versões antigas.
  const now=Date.now();
  if(!force&&now-hlgbRecordLastPullAt<8000)return false;
  hlgbRecordLastPullAt=now;hlgbRecordPulling=true;
  try{
    // Guardamos o último instante realmente conhecido ANTES da leitura. A consulta usa 1 s de
    // sobreposição para não perder registros, então ela pode devolver novamente linhas já aplicadas.
    // Essas linhas repetidas não podem provocar renderAll(), senão a tela dá a impressão de tremida.
    const lastBeforePull=String(hlgbRecordGlobalLastSeen||"");
    let since=null;
    if(lastBeforePull){
      const t=Date.parse(lastBeforePull);
      if(Number.isFinite(t))since=new Date(Math.max(0,t-1000)).toISOString();
    }
    const bundle=await hlgbRecordLoadBundle({preserveLocal:true,since});
    const rows=Array.isArray(bundle?.rows)?bundle.rows:[];
    const freshRows=lastBeforePull
      ? rows.filter(r=>String(r?.updated_at||"")>lastBeforePull)
      : rows;
    const changed=freshRows.length>0;
    if(changed){
      localSaveOnly();
      const modules=[...new Set(freshRows.map(r=>String(r?.module||"")).filter(Boolean))];
      // Produtos e Folha atualizam a tela mesmo com filtros focados.
      let handled=false;
      if(modules.includes("products"))handled=hlgbRenderIncomingRecord("products")||handled;
      if(modules.includes("payroll"))handled=hlgbRenderIncomingRecord("payroll")||handled;
      if(!handled&&!cloudUserIsEditing()){
        const prev=cloudApplying;cloudApplying=true;try{renderAll();cloudRemoteUpdatePending=false}finally{cloudApplying=prev}
      }else if(!handled) cloudRemoteUpdatePending=true;
    }
    return changed;
  }catch(e){
    // Uma falha transitória de leitura jamais desativa o modo por registros.
    console.warn("HLGB v91 pull bundle",e);return false;
  }finally{hlgbRecordPulling=false}
}

function hlgb928EnsureCloudCoverage(module){
  try{
    if(!HLGB_RECORD_MODULES.includes(module))return 0;
    const snap=hlgbRecordSnapshots?.[module];
    if(!(snap instanceof Map))return 0;
    const local=hlgbRecordLocalMap(module);let added=0;
    for(const [id,row] of snap.entries()){
      if(!row||row.deleted_at||local.has(String(id)))continue;
      const data=hlgbRecordClone(row.data);
      if(data&&typeof data==="object"){
        local.set(String(id),data);added++;
      }
    }
    if(added)db[module]=[...local.values()];
    return added;
  }catch(e){console.warn("HLGB v91.28 coverage",module,e);return 0}
}

function cutPriorityWeight(o){
  let p=String(o?.priority||"Padrão").normalize("NFD").replace(/[\u0300-\u036f]/g,"").toLowerCase();
  if(p.includes("urgentissimo"))return 0;
  if(p.includes("urgente"))return 1;
  return 2;
}
function cloneCutGrade(v){try{return JSON.parse(JSON.stringify(v||[]))}catch(e){return []}}
function prepareCutSchedule(){
  db.cuts=Array.isArray(db.cuts)?db.cuts:[];
  let today=isoDate(new Date()),changed=false;
  let orders=pedidosQuePrecisamDeCorte();
  orders.forEach(o=>{
    let c=obterCorteDoPedido(o);
    let w=cutPriorityWeight(o);
    // Urgente/urgentíssimo sem programação entra automaticamente no dia atual.
    if(!c.plannedCutDate && w<=1){c.plannedCutDate=today;c.autoScheduledPriority=true;changed=true}
    // Tudo que ficou para trás e não foi finalizado passa para o dia atual.
    if(String(c.status||"").toLowerCase()!=="finalizado" && c.plannedCutDate && c.plannedCutDate<today){
      c.carriedFromDate=c.carriedFromDate||c.plannedCutDate;
      c.lastCarriedFrom=c.plannedCutDate;
      c.plannedCutDate=today;
      c.rolloverCount=(+c.rolloverCount||0)+1;
      changed=true;
    }
  });
  if(changed){try{persistDb()}catch(e){console.error(e)}}
}

function cutOriginalGrade(c,o){
  if(Array.isArray(c.originalGrade)&&c.originalGrade.length)return c.originalGrade;
  return cloneCutGrade(o?.grade||[]);
}
function cutActualGrade(c,o){
  if(Array.isArray(c.actualCutGrade)&&c.actualCutGrade.length)return c.actualCutGrade;
  return cutOriginalGrade(c,o);
}
function cutGradeComparisonHTML(c){
  let o=cutOrderFor(c),orig=cutOriginalGrade(c,o),actual=cutActualGrade(c,o);
  if(!orig.length)return `<div class="panel"><strong>Pedido original:</strong> ${(+c.originalPieces||qtyOfOrder(o||{})||0).toLocaleString("pt-BR")} peças<br><strong>Cortado:</strong> ${(+c.pieces||0).toLocaleString("pt-BR")} peças</div>`;
  let byKey={};
  const key=x=>`${x.productId||""}|${x.color||""}|${x.size||""}`;
  orig.forEach(x=>byKey[key(x)]={orig:+x.qty||0,actual:0,item:x});
  actual.forEach(x=>{let k=key(x);if(!byKey[k])byKey[k]={orig:0,actual:0,item:x};byKey[k].actual+=+x.qty||0});
  let rows=Object.values(byKey).map(r=>{
    let p=(db.products||[]).find(x=>+x.id===+r.item.productId),diff=r.actual-r.orig;
    return [esc(p?.name||"Produto"),esc(r.item.color||"-"),esc(r.item.size||"-"),r.orig,r.actual,diff===0?"0":`<span class="${diff<0?'cut-diff-negative':'cut-diff-positive'}">${diff>0?'+':''}${diff}</span>`];
  });
  return table(["Produto","Cor","Tamanho","Pedido original","Cortado","Diferença"],rows)+`<div class="sub" style="margin-top:8px"><strong>Observação:</strong> ${esc(c.cutAdjustmentNote||"-")}</div>`;
}
